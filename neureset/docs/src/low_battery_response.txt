@startuml
actor        operator as op
participant  Headset as hd
participant  Device as dev
participant  MainWindow as main
participant  DeviceBattery as bat

bat -> dev : lowBattery()
note left
Low battery is enough 
to finish any ongoing treatments
end note
activate dev
dev -> dev : lowBatteryNotification()
dev -> dev : lowBatteryIcon()


alt Battery is replaced
  op -> dev : press "Power button"
  dev -> dev : turnOff()
  op -> dev : manually replace battery
  op -> dev : press "Power button"
  dev -> dev : turnOn()
  note right
  normal operation resumes
  end note

else Battery runs out completly
  op -> dev : press "New session"
  dev -> dev: lowBatteryNotification()
  note right
  device does not start a new session
 while low on battery
  end note
  bat -> dev: noBattery()
  dev -> dev : turnOff()
end


@enduml
